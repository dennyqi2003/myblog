$$
\begin{aligned}&(A \land B)\to C\\ \iff & \neg( A\land B)\lor C \\ \iff & (\neg A \lor \neg B) \lor C \\ \iff & \neg A \lor (\neg B \lor C) \\ \iff & A \to (B \to C)\\ \iff & A\to B \to C & \text{(imply是右结合的)}\end{aligned}
$$
